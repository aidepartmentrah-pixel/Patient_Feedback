### Some Extra Points on the View
**Creating the view**

```sql
CREATE VIEW vw_IncidentFull
AS
SELECT
    -- =========================
    -- Incident (Request level)
    -- =========================
    IR.UniqueID                    AS IncidentRequestID,
    IR.Code                        AS IncidentCode,
    IR.PatientName,
    IR.PatientID,
    IR.MRN,
    IR.DateAndTimeRecieved         AS ReceivedAt,
    IR.IsInPatient,
    IR.IncidentStatusID,
    IR.IssuingOrgUnitID,

    -- =========================
    -- Case level
    -- =========================
    IRC.UniqueID                   AS IncidentCaseID,
    IRC.Description                AS CaseDescription,
    IRC.DateAndTimeCreated         AS CaseCreatedAt,
    IRC.DateAndTimeHappened,
    IRC.IncidentCaseSeverityID,
    IRC.IncidentCaseDomainID,
    IRC.IncidentCaseStageID,
    IRC.IncidentCaseHarmID,
    IRC.DepartmentExplanationStatusID,

    -- =========================
    -- Lookup expansions (readable)
    -- =========================
    D.DomainName                   AS DomainName,
    S.StageName                    AS StageName,
    H.HarmLevel                    AS HarmLevel,
    DES.StatusName                 AS DepartmentExplanationStatus,

    -- =========================
    -- Latest Action (if exists)
    -- =========================
    A.UniqueID                     AS LatestActionID,
    A.Description                  AS LatestActionDescription,
    A.DateAndTimeSubmit            AS ActionSubmittedAt,
    A.DateAndTimeDepartmentNoteSaved,

    -- =========================
    -- Flags
    -- =========================
    CASE 
        WHEN A.DateAndTimeSubmit IS NOT NULL THEN 1 
        ELSE 0 
    END                            AS HasAction,

    CASE 
        WHEN A.DateAndTimeSubmit IS NOT NULL 
             AND A.DateAndTimeSubmit < GETDATE() THEN 1
        ELSE 0
    END                            AS IsOverdue

FROM IncidentRequest IR

LEFT JOIN IncidentRequestCase IRC
    ON IRC.IncidentRequestID = IR.UniqueID

-- Latest action per case
LEFT JOIN IncidentRequestCaseAction A
    ON A.IncidentRequestCaseID = IRC.UniqueID
   AND A.DateAndTimeCreated = (
        SELECT MAX(A2.DateAndTimeCreated)
        FROM IncidentRequestCaseAction A2
        WHERE A2.IncidentRequestCaseID = IRC.UniqueID
   )

-- Lookups
LEFT JOIN IncidentCaseDomain D
    ON D.DomainID = IRC.IncidentCaseDomainID

LEFT JOIN IncidentCaseStage S
    ON S.StageID = IRC.IncidentCaseStageID

LEFT JOIN IncidentCaseHarm H
    ON H.HarmID = IRC.IncidentCaseHarmID

LEFT JOIN DepartmentExplanationStatus DES
    ON DES.StatusID = IRC.DepartmentExplanationStatusID;

```
